uvicorn[standard]

# --- Motor fiscal (o que o spike provou em produção) ---

signxml         # assina XMLDSig (armadilha 4: sem prefixo de namespace)

cryptography    # PKCS#12 (.pfx) em memória + AES-256-GCM da custódia (KEK/DEK)
